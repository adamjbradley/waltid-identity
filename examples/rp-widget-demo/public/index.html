Quick Integration Guide

            
Add age verification to your site in minutes


            

                
Step 1: Include the SDK

                

                    <!-- Add this to your HTML -->

                    <script src="https://verify.example.com/widget/v1/sdk.js"></script>
                

            


            

                
Step 2: Get a Client Token (Server-side)

                

                    // Your backend generates a token using your API key

                    const response = await fetch('/api/token');

                    const { clientToken } = await response.json();
                

            


            

                
Step 3: Initialize and Verify

                

                    // Initialize the SDK

                    WaltVerify.init({ clientToken: clientToken });

                    

                    // Age verification (modal mode)

                    WaltVerify.verifyAge({

                      minAge: 18,

                      onSuccess: (result) => {

                        console.log('User is verified!', result);

                      },

                      onFailure: (error) => {

                        console.log('Verification failed', error);

                      }

                    });

                    

                    // Or use inline mode

                    WaltVerify.verify({

                      template: 'kyc_basic',

                      ui: 'inline',

                      container: '#verification-container'

                    });
